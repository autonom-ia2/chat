<!-- Botão do WhatsApp que avisa o clique ao Chat2You -->
<script>
(function () {
  var SIGNAL_URL = %{signal_url};
  var WHATSAPP_URL = %{whatsapp_url};
  var MESSAGE = %{message};
  var ALPHABET = %{alphabet};
  var KEYS = ["utm_source", "utm_medium", "utm_campaign", "utm_term", "utm_content", "utm_id", "fbclid"];
  var STORE = "chat2you_attribution";

  function remember() {
    try {
      var query = new URLSearchParams(window.location.search);
      var found = {};
      var any = false;
      KEYS.forEach(function (key) {
        var value = query.get(key);
        if (value) { found[key] = value.slice(0, 512); any = true; }
      });
      if (any) sessionStorage.setItem(STORE, JSON.stringify(found));
    } catch (error) {}
  }

  function remembered() {
    try { return JSON.parse(sessionStorage.getItem(STORE) || "{}"); } catch (error) { return {}; }
  }

  function newToken() {
    var bytes = new Uint8Array(8);
    window.crypto.getRandomValues(bytes);
    return Array.prototype.map.call(bytes, function (byte) { return ALPHABET[byte %% ALPHABET.length]; }).join("");
  }

  function report(token) {
    var body = JSON.stringify({
      token: token,
      page_url: window.location.origin + window.location.pathname,
      consent: false,
      params: remembered(),
      lead: { fields: [] }
    });
    try {
      if (navigator.sendBeacon && navigator.sendBeacon(SIGNAL_URL, new Blob([body], { type: "text/plain" }))) return;
      fetch(SIGNAL_URL, { method: "POST", mode: "no-cors", keepalive: true, headers: { "Content-Type": "text/plain" }, body: body });
    } catch (error) {}
  }

  remember();
  document.addEventListener("click", function (event) {
    var button = event.target.closest("[data-chat2you-whatsapp]");
    if (!button) return;
    event.preventDefault();
    var token = newToken();
    report(token);
    window.open(WHATSAPP_URL + encodeURIComponent(MESSAGE + " #" + token), "_blank", "noopener");
  });
})();
</script>
