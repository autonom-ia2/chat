    it 'adversarial lock proof: concurrent UPDATE waits for the processor row lock' do
      follow_up = reminder_setup
      started = Queue.new
      worker = nil
      follow_up.with_lock do
        worker = Thread.new do
          ActiveRecord::Base.connection_pool.with_connection do |connection|
            row = Crm::FollowUp.find(follow_up.id)
            started << connection.select_value('SELECT pg_backend_pid()')
            row.update!(status: :canceled)
          end
        end
        pid = started.pop
        waiting = nil
        Timeout.timeout(5) do
          loop do
            waiting = ActiveRecord::Base.connection.select_value("SELECT wait_event_type FROM pg_stat_activity WHERE pid = #{Integer(pid)}")
            break if waiting == 'Lock'

            sleep 0.01
          end
        end
        expect(waiting).to eq('Lock')
        expect(follow_up.reload).to be_pending
      end
      expect(worker.join(5)).not_to be_nil
      expect(follow_up.reload).to be_canceled
    ensure
      worker&.join(5)
    end
