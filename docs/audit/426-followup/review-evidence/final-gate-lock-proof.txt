    %w[card pipeline].each do |target|
      it "final gate lock proof: concurrent #{target} update waits for decision" do
        follow_up = reminder_setup
        runner = described_class.new(follow_up: follow_up, now: now)
        worker = nil
        allow(runner).to receive(:persist_decision).and_wrap_original do |method, payload|
          started = Queue.new
          worker = Thread.new do
            ActiveRecord::Base.connection_pool.with_connection do |connection|
              row = target == 'card' ? Crm::Card.find(follow_up.card_id) : Crm::Pipeline.find(follow_up.card.pipeline_id)
              started << connection.select_value('SELECT pg_backend_pid()')
              if target == 'card'
                row.update!(status: :won)
              else
                data = row.metadata.deep_dup
                data['ai']['auto_followup']['allowed_days'] = [2]
                row.update!(metadata: data)
              end
            end
          end
          pid = started.pop
          waiting = nil
          Timeout.timeout(5) do
            loop do
              waiting = ActiveRecord::Base.connection.select_value("SELECT wait_event_type FROM pg_stat_activity WHERE pid = #{Integer(pid)}")
              break if waiting == 'Lock'

              sleep 0.01
            end
          end
          expect(waiting).to eq('Lock')
          method.call(payload)
        end
        expect(runner.perform.status).to eq(:reminded)
        expect(worker.join(5)).not_to be_nil
        if target == 'card'
          expect(follow_up.card.reload).to be_won
        else
          expect(follow_up.card.pipeline.reload.metadata.dig('ai', 'auto_followup', 'allowed_days')).to eq([2])
        end
      ensure
        worker&.join(5)
      end
    end
